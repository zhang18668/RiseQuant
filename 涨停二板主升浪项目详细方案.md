**F-001: 基础量价因子**

```python
# feature/price_volume.py
class PriceVolumeFactors:
    """量价因子计算器"""

    @staticmethod
    def returns(close: pd.Series, n: int) -> pd.Series:
        """N日收益率"""
        return close.pct_change(n)

    @staticmethod
    def volume_ratio(volume: pd.Series, n: int = 5) -> pd.Series:
        """量比 = 成交量 / N日均量"""
        return volume / volume.rolling(n).mean()

    @staticmethod
    def volatility(close: pd.Series, n: int = 5) -> pd.Series:
        """N日波动率"""
        return close.pct_change().rolling(n).std()

    @staticmethod
    def ma_position(close: pd.Series, n: int) -> pd.Series:
        """收盘价 / N日均线位置"""
        ma = close.rolling(n).mean()
        return close / ma

    @staticmethod
    def amplitude(high: pd.Series, low: pd.Series, n: int = 5) -> pd.Series:
        """N日振幅"""
        return (high.rolling(n).max() / low.rolling(n).min()) - 1
```
